# icons for the dark mode: same prefix and names (nCore keeps only one of the two resources at startup).
# The dark png are generated at build time in ${CMAKE_CURRENT_BINARY_DIR}/icons_dark from the svg of the light icons
# (resources/icons/darkify.py + rsvg-convert or inkscape); the light icon is used when there is no svg,
# or when python or the svg renderer are not found.
# Dark icons of the plugins: the plugins compile only their light icons (same /icons/ names),
# these are registered by Neutrino before any plugin is loaded and the first registered resource wins
# (plugins/Lineout/Lineout_H: one level deeper; refresh2.png is in Ghost_fringes and Visar: the first one is used)
find_program(NEUTRINO_PYTHON NAMES python3 python)
# find_program(NEUTRINO_SVG_RENDERER NAMES rsvg-convert inkscape PATHS /Applications/Inkscape.app/Contents/MacOS)
# rsvg-convert first (on macOS find_program looks in the app bundles first, i.e. it found Inkscape.app)
find_program(NEUTRINO_SVG_RENDERER NAMES rsvg-convert)
if(NOT NEUTRINO_SVG_RENDERER)
    find_program(NEUTRINO_SVG_RENDERER NAMES inkscape PATHS /Applications/Inkscape.app/Contents/MacOS)
endif()
set(DARKIFY ${ICONSDIR}/darkify.py)
set(DARK_ICONS_DIR ${CMAKE_CURRENT_BINARY_DIR}/icons_dark)
if(NEUTRINO_PYTHON AND NEUTRINO_SVG_RENDERER)
    message(STATUS "Dark icons: ${NEUTRINO_PYTHON} ${DARKIFY} + ${NEUTRINO_SVG_RENDERER}")
    set(MAKE_DARK_ICONS ON)
else()
    message(WARNING "Dark icons: python3 or rsvg-convert (librsvg) not found, the dark mode will use the light icons")
    set(MAKE_DARK_ICONS OFF)
endif()

file(GLOB PLUGINS_ICONS CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/../plugins/*/*.png ${CMAKE_CURRENT_SOURCE_DIR}/../plugins/*/*/*.png)

set(QRC_CONTENT "<RCC>\n    <qresource prefix=\"/icons/\">\n")
set(DARK_ALIASES)
set(DARK_PNGS)
foreach(my_file ${ICONS} ${PLUGINS_ICONS})
    get_filename_component(my_file_basename ${my_file} NAME)
    get_filename_component(my_file_dir ${my_file} DIRECTORY)
    get_filename_component(my_file_name ${my_file} NAME_WE)
    set(my_svg ${my_file_dir}/${my_file_name}.svg)
    set(is_plugin OFF)
    if(my_file IN_LIST PLUGINS_ICONS)
        set(is_plugin ON)
    endif()
    if(my_file_basename IN_LIST DARK_ALIASES)
        continue()
    endif()
    if(MAKE_DARK_ICONS AND EXISTS ${my_svg})
        set(my_dark ${DARK_ICONS_DIR}/${my_file_basename})
        add_custom_command(OUTPUT ${my_dark}
            COMMAND ${CMAKE_COMMAND} -E make_directory ${DARK_ICONS_DIR}
            COMMAND ${NEUTRINO_PYTHON} ${DARKIFY} ${my_svg} ${DARK_ICONS_DIR}/${my_file_name}.svg
            COMMAND ${NEUTRINO_SVG_RENDERER} ${DARK_ICONS_DIR}/${my_file_name}.svg -o ${my_dark}
            DEPENDS ${my_svg} ${DARKIFY}
            COMMENT "Dark icon ${my_file_basename}"
            VERBATIM)
        list(APPEND DARK_PNGS ${my_dark})
        set(my_file ${my_dark})
    elseif(is_plugin)
        continue() # plugin icon without svg: the light one, compiled in the plugin
    endif()
    file(RELATIVE_PATH my_file_relative_path ${QRCDIR} ${my_file})
    string(APPEND QRC_CONTENT "        <file alias=\"${my_file_basename}\">${my_file_relative_path}</file>\n")
    list(APPEND DARK_ALIASES ${my_file_basename})
endforeach()
string(APPEND QRC_CONTENT "    </qresource>\n</RCC>")
file(CONFIGURE OUTPUT ${ICONS_DARK_QRC} CONTENT "${QRC_CONTENT}" @ONLY)
